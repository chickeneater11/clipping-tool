cmake_minimum_required(VERSION 3.20)
project(clipping_tool VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(clipping_core INTERFACE)
target_include_directories(clipping_core INTERFACE include)
if(MSVC)
  target_compile_options(clipping_core INTERFACE /W4 /WX /permissive-)
else()
  target_compile_options(clipping_core INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(clipping_probe src/main.cpp)
target_link_libraries(clipping_probe PRIVATE clipping_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(replay_buffer_tests tests/replay_buffer_tests.cpp)
  target_link_libraries(replay_buffer_tests PRIVATE clipping_core)
  add_test(NAME replay_buffer COMMAND replay_buffer_tests)
endif()
