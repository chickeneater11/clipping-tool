cmake_minimum_required(VERSION 3.20)
project(clipping_tool VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(clipping_core INTERFACE)
target_include_directories(clipping_core INTERFACE include)
if(MSVC)
  target_compile_options(clipping_core INTERFACE /W4 /WX /permissive-)
else()
  target_compile_options(clipping_core INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(clipping_probe src/main.cpp)
target_link_libraries(clipping_probe PRIVATE clipping_core)
if(WIN32)
  add_executable(clipping_capabilities src/windows/capability_probe.cpp)
  target_compile_definitions(clipping_capabilities PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(clipping_capabilities PRIVATE clipping_core d3d11 dxgi mfplat mf mfuuid ole32)
  add_executable(clipping_record_test src/windows/record_test.cpp)
  target_compile_definitions(clipping_record_test PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(clipping_record_test PRIVATE clipping_core d3d11 dxgi mfplat mf mfuuid mfreadwrite ole32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(replay_buffer_tests tests/replay_buffer_tests.cpp)
  target_link_libraries(replay_buffer_tests PRIVATE clipping_core)
  add_test(NAME replay_buffer COMMAND replay_buffer_tests)
  if(WIN32)
    add_test(NAME recorder_help COMMAND clipping_record_test --help)
    set_tests_properties(recorder_help PROPERTIES PASS_REGULAR_EXPRESSION "Usage: clipping_record_test")
    add_test(NAME recorder_invalid_duration COMMAND clipping_record_test unused.mp4 0)
    set_tests_properties(recorder_invalid_duration PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
